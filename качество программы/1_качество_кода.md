# Анализ качества кода проекта Bukings

## Общая оценка: 6/10

Проект демонстрирует базовое понимание принципов разработки, но имеет существенные проблемы в области качества кода, которые требуют внимания.

---

## 1. ПОЛОЖИТЕЛЬНЫЕ СТОРОНЫ

### 1.1 Модульная структура
✅ **Хорошо**: Код разделен на логические модули:
- `backend/` - бизнес-логика
- `middleware/` - промежуточные обработчики
- `db_local/` - работа с базой данных
- `public/` - клиентская часть

### 1.2 Использование современных технологий
✅ **Хорошо**: 
- Express.js для API
- PostgreSQL для БД
- JWT для аутентификации
- bcrypt для хеширования паролей

### 1.3 Валидация данных
✅ **Хорошо**: Реализованы middleware для валидации:
```javascript
// middleware/validators.js
const validateRegister = (req, res, next) => {
  // Проверка email, пароля, имени
}
```

### 1.4 Обработка ошибок
✅ **Хорошо**: Централизованная обработка через `ApiError` класс

## 2. КРИТИЧЕСКИЕ ПРОБЛЕМЫ

### 2.1 Дублирование кода (HIGH)

**Проблема**: Повторяющиеся паттерны в `api.js`

**Примеры**:
```javascript
// Повторяется 6 раз!
app.get('/api/users', async (req, res) => {
    if (!con._connected) {
        throw new ApiError(500, 'Database connection not established');
    }
    try {
        const result = await con.query('SELECT * FROM users');
        res.json(result.rows);
    } catch (error) {
        throw new ApiError(500, 'Ошибка сервера');
    }
});
```

**Исправление**:
```javascript
// backend/crud_generator.js
function generateCRUDHandlers(table, publicFields = '*') {
    return {
        getAll: async (req, res, next) => {
            try {
                const result = await con.query(`SELECT ${publicFields} FROM ${table}`);
                res.json(result.rows);
            } catch (error) {
                next(new ApiError(500, 'Ошибка сервера'));
            }
        },
        // ... getById, update, delete
    };
}

// api.js
const userHandlers = generateCRUDHandlers('users', 'id, name, email, role_id, created_at');
app.get('/api/users', userHandlers.getAll);
```

**Выгода**: 
- Уменьшение кода на ~60%
- Единая точка изменений
- Меньше ошибок

### 2.2 Смешивание логики и роутинга (MEDIUM)

**Проблема**: `server.js` содержит и роутинг, и бизнес-логику

**Текущий код** (server.js, строки 24-46):
```javascript
app.post('/api/users', validateRegister, async (req, res, next) => {
    const { name, email, password, role_id } = req.body;
    if (!name || !email || !password) {
        return res.status(400).json({ error: 'Заполните все поля' });
    }
    try {
        const check = await con.query('SELECT 1 FROM users WHERE email = $1', [email]);
        if (check.rowCount > 0) {
            return res.status(409).json({ error: 'Пользователь с таким email уже существует' });
        }
        const hashed = await hashPassword(password);
        const result = await con.query(
            'INSERT INTO users (name, email, password_hash, role_id, created_at) VALUES ($1, $2, $3, $4, NOW()) RETURNING id, name, email, role_id',
            [name, email, hashed, role_id || 1]
        );
        res.status(201).json(result.rows[0]);
    } catch (error) {
        console.error('Ошибка при добавлении пользователя', error);
        next(new ApiError(500, 'Ошибка сервера'));
    }
});
```

**Исправление**:
```javascript
// backend/services/userService.js
class UserService {
    async createUser(userData) {
        const { name, email, password, role_id = 1 } = userData;
        
        // Проверка дубликатов
        const existing = await con.query('SELECT 1 FROM users WHERE email = $1', [email]);
        if (existing.rowCount > 0) {
            throw new ApiError(409, 'Пользователь с таким email уже существует');
        }
        
        // Хеширование пароля
        const password_hash = await hashPassword(password);
        
        // Создание пользователя
        const result = await con.query(
            'INSERT INTO users (name, email, password_hash, role_id, created_at) VALUES ($1, $2, $3, $4, NOW()) RETURNING id, name, email, role_id',
            [name, email, password_hash, role_id]
        );
        
        return result.rows[0];
    }
}

// routes/userRoutes.js
const userService = new UserService();

router.post('/', validateRegister, async (req, res, next) => {
    try {
        const user = await userService.createUser(req.body);
        res.status(201).json(user);
    } catch (error) {
        next(error);
    }
});

// server.js
app.use('/api/users', userRoutes);
```

**Выгода**:
- Чистое разделение ответственности
- Легче тестировать бизнес-логику
- Переиспользование сервисов

// Установить: npm install winston

const winston = require('winston');

const logger = winston.createLogger({
    level: process.env.LOG_LEVEL || 'info',
    format: winston.format.combine(
        winston.format.timestamp(),
        winston.format.errors({ stack: true }),
        winston.format.json()

    transports: [
        new winston.transports.File({ filename: 'logs/error.log', level: 'error' }),

if (process.env.NODE_ENV !== 'production') {
    logger.add(new winston.transports.Console({
        format: winston.format.combine(
            winston.format.colorize(),
            winston.format.simple()
        )
    }));

module.exports = logger;

// Использование

## 3. ПРОБЛЕМЫ АРХИТЕКТУРЫ

### 3.1 Отсутствие слоя репозиториев

**Проблема**: Прямые запросы к БД в роутах

**Рекомендация**: Создать Data Access Layer

```javascript
// repositories/UserRepository.js
class UserRepository {
    constructor(dbConnection) {
        this.db = dbConnection;
    }
    
    async findByEmail(email) {
        const result = await this.db.query(
            'SELECT * FROM users WHERE email = $1',
            [email]
        );
        return result.rows[0];
    }
    
    async create(userData) {
        const { name, email, password_hash, role_id } = userData;
        const result = await this.db.query(
            'INSERT INTO users (name, email, password_hash, role_id, created_at) VALUES ($1, $2, $3, $4, NOW()) RETURNING *',
            [name, email, password_hash, role_id]
        );
        return result.rows[0];
    }
    
    async findById(id) {
        const result = await this.db.query(
            'SELECT id, name, email, role_id, created_at FROM users WHERE id = $1',
            [id]
        );
        return result.rows[0];
    }
}

module.exports = UserRepository;
```

### 3.2 Отсутствие конфигурации окружений

**Проблема**: Нет валидации переменных окружения

**Рекомендация**:
```javascript
// config/env.js
const requiredEnvVars = [
    'DB_HOST',
    'DB_USER', 
    'DB_PASSWORD',
    'DB_NAME',
    'JWT_SECRET'
];

function validateEnv() {
    const missing = requiredEnvVars.filter(key => !process.env[key]);
    
    if (missing.length > 0) {
        throw new Error(`Missing required environment variables: ${missing.join(', ')}`);
    }
}

module.exports = {
    validateEnv,
    db: {
        host: process.env.DB_HOST,
        user: process.env.DB_USER,
        password: process.env.DB_PASSWORD,
        database: process.env.DB_NAME,
        port: parseInt(process.env.DB_PORT || '5432', 10),
    },
    jwt: {
        secret: process.env.JWT_SECRET,
        expiresIn: process.env.JWT_EXPIRES_IN || '24h',
    },
    server: {
        port: parseInt(process.env.SERVER_PORT || '3000', 10),
        nodeEnv: process.env.NODE_ENV || 'development',
    }
};

// server.js
const config = require('./config/env');
config.validateEnv();
```
