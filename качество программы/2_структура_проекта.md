**Проблема**: Вся логика в роутах

**Текущий код** (api.js):
```javascript

    const { id } = req.params;
    try {
        // ❌ Проверка прав - это логика контроллера
        const user = await con.query('SELECT role_id FROM users WHERE id = $1', [req.user.id]);
        if (user.rowCount === 0 || user.rows[0].role_id !== 3) {
            return res.status(403).json({ error: 'Нет прав' });
        }

        // ❌ Работа с БД - это логика репозитория
        const result = await buildDelete('users', id);
        if (result.rowCount === 0) {
            return res.status(404).json({ error: 'Пользователь не найден' });
        }
        
        res.json({ message: 'Пользователь удалён', user: result.rows[0] });
    } catch (error) {
        next(new ApiError(500, 'Ошибка сервера'));
    }

```

**Правильная архитектура**:

```javascript
// repositories/UserRepository.js
class UserRepository {
    async findById(id) {
        const result = await db.query('SELECT * FROM users WHERE id = $1', [id]);
        return result.rows[0];
    }
    
    async delete(id) {
        const result = await db.query('DELETE FROM users WHERE id = $1 RETURNING *', [id]);
        return result.rows[0];
    }
}

// services/UserService.js
class UserService {
    constructor(userRepository) {
        this.userRepository = userRepository;
    }
    
    async deleteUser(id, currentUserId) {
        // Проверка прав
        const currentUser = await this.userRepository.findById(currentUserId);
        if (!currentUser || currentUser.role_id !== ROLES.ADMIN) {
            throw new ApiError(403, 'Нет прав для удаления пользователя');
        }
        
        // Удаление
        const deletedUser = await this.userRepository.delete(id);
        if (!deletedUser) {
            throw new ApiError(404, 'Пользователь не найден');
        }
        
        return deletedUser;
    }
}

// controllers/UserController.js
class UserController {
    constructor(userService) {
        this.userService = userService;
    }
    
    async deleteUser(req, res, next) {
        try {
            const { id } = req.params;
            const deletedUser = await this.userService.deleteUser(id, req.user.id);
            res.json({ message: 'Пользователь удалён', user: deletedUser });
        } catch (error) {
            next(error);
        }
    }
}

// routes/user.routes.js
const router = express.Router();
const userController = new UserController(userService);

router.delete('/:id', verifyToken, (req, res, next) => userController.deleteUser(req, res, next));

// app.js
app.use('/api/users', userRoutes);
```

**Исправление**:
```javascript
// config/env.js
const dotenv = require('dotenv');
const path = require('path');

// Загрузка .env файла
dotenv.config({ path: path.join(__dirname, '../.env') });

// Обязательные переменные
const required = ['DB_HOST', 'DB_USER', 'DB_PASSWORD', 'DB_NAME', 'JWT_SECRET'];

// Валидация
function validateEnv() {
    const missing = required.filter(key => !process.env[key]);
    
    if (missing.length > 0) {
        throw new Error(
            `❌ Missing required environment variables:\n` +
            missing.map(key => `  - ${key}`).join('\n') +
            `\n\nCreate a .env file based on .env.example`
        );
    }
    
    console.log('✅ Environment variables validated');
}

module.exports = {
    validateEnv,
    env: process.env.NODE_ENV || 'development',
    port: parseInt(process.env.PORT || '3000', 10),
    database: {
        host: process.env.DB_HOST,
        port: parseInt(process.env.DB_PORT || '5432', 10),
        user: process.env.DB_USER,
        password: process.env.DB_PASSWORD,
        database: process.env.DB_NAME,
    },
    jwt: {
        secret: process.env.JWT_SECRET,
        expiresIn: process.env.JWT_EXPIRES_IN || '24h',
    },
};

// server.js
const config = require('./config/env');
config.validateEnv();  // Проверка перед запуском
```

```
## Архитектура

Проект следует 3-tier архитектуре:

1. **Presentation Layer** (Controllers + Routes)
   - Обработка HTTP запросов/ответов
   - Валидация входных данных
   - Сериализация ответов

2. **Business Logic Layer** (Services)
   - Бизнес-правила
   - Обработка данных
   - Координация между репозиториями

3. **Data Access Layer** (Repositories)
   - Работа с базой данных
   - SQL запросы
   - Маппинг данных

## Поток данных

```
Request → Route → Middleware → Controller → Service → Repository → Database
                                                                        ↓
Response ← Route ← Middleware ← Controller ← Service ← Repository ← Database
