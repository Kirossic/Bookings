## 1. КРИТИЧЕСКИЕ УЯЗВИМОСТИ (CRITICAL) 🔴

### 1.1 SQL Injection через название таблиц

**Серьезность**: CRITICAL  
**Файл**: `backend/api_constructor.js`  
**Строки**: 4, 22, 28

**Проблема**: Название таблицы вставляется через строковую интерполяцию

```javascript
// ❌ УЯЗВИМО!
async function buildRead(table, id) {
    const query = `SELECT * FROM ${table} WHERE id = $1`;  // table не экранируется!
    const values = [id];
    return await con.query(query, values);
}

async function buildUpdate(table, id, data, allowed = []) {
    // ...
    const query = `UPDATE ${table} SET ${updates.join(', ')} WHERE id = $${index} RETURNING *`;
    // ...
}

async function buildDelete(table, id) {
    const query = `DELETE FROM ${table} WHERE id = $1 RETURNING *`;
    // ...
}
```

**Эксплойт**:
```javascript
// Злоумышленник может вызвать:
await buildRead("users; DROP TABLE users; --", 1);

// Результирующий SQL:
// SELECT * FROM users; DROP TABLE users; -- WHERE id = $1
```

**Исправление 1**: Whitelist разрешенных таблиц
```javascript
const ALLOWED_TABLES = ['users', 'resources', 'bookings', 'payments', 'roles', 'logs'];

async function buildRead(table, id) {
    // Валидация таблицы
    if (!ALLOWED_TABLES.includes(table)) {
        throw new ApiError(400, 'Invalid table name');
    }
    
    const query = `SELECT * FROM ${table} WHERE id = $1`;
    const values = [id];
    return await con.query(query, values);
}
```

**Исправление 2**: Использовать pg-format для безопасного экранирования
```javascript
const format = require('pg-format');

async function buildRead(table, id) {
    // pg-format автоматически экранирует идентификаторы
    const query = format('SELECT * FROM %I WHERE id = $1', table);
    const values = [id];
    return await con.query(query, values);
}
```

**Исправление 3**: Создать отдельный репозиторий для каждой таблицы (РЕКОМЕНДУЕТСЯ)
```javascript
// repositories/UserRepository.js
class UserRepository {
    async findById(id) {
        // Название таблицы зашито в коде
        const result = await db.query('SELECT * FROM users WHERE id = $1', [id]);
        return result.rows[0];
    }
}
```

### 1.2 Hardcoded JWT Secret

**Серьезность**: CRITICAL  
**Файл**: `backend/Verify_Token.js`, `api.js`  
**Строки**: 2, 32

**Проблема**: JWT подписывается с дефолтным секретом

```javascript
// ❌ КРИТИЧЕСКАЯ УЯЗВИМОСТЬ!
const JWT_SECRET = process.env.JWT_SECRET || '789456123';

const token = jwt.sign(
    { id: user.id, email: user.email, role_id: user.role_id },
    process.env.JWT_SECRET || '789456123',  // Дефолтное значение!
    { expiresIn: '24h' }
);
```

**Риск**:
- ✅ Злоумышленник знает секрет → может создавать валидные токены
- ✅ Может авторизоваться как любой пользователь (включая админа)
- ✅ Полная компрометация системы аутентификации

**Исправление**:
```javascript
// config/env.js
function validateEnv() {
    if (!process.env.JWT_SECRET) {
        throw new Error(
            '❌ CRITICAL: JWT_SECRET is not set!\n' +
            'Generate a strong secret: openssl rand -base64 32'
        );
    }
    
    if (process.env.JWT_SECRET.length < 32) {
        throw new Error('❌ JWT_SECRET must be at least 32 characters long');
    }
}

// backend/Verify_Token.js
const config = require('../config/env');

function verifyToken(req, res, next) {
    const authHeader = req.headers['authorization'];
    const token = authHeader && authHeader.split(' ')[1];

    if (!token) {
        return res.status(401).json({ error: 'Токен отсутствует' });
    }

    jwt.verify(token, config.jwt.secret, (err, user) => {  // НЕТ дефолтного значения!
        if (err) {
            return res.status(403).json({ error: 'Недействительный токен' });
        }
        req.user = user;
        next();
    });
}
```

**Генерация безопасного секрета**:
```bash
# В терминале
openssl rand -base64 64

# Добавить в .env
JWT_SECRET=ваш_сгенерированный_секрет_длиной_минимум_64_символа
```

### 1.4 Автоматическое удаление дубликатов при подключении

**Серьезность**: HIGH  
**Файл**: `db_local/db.js`  
**Строки**: 17-32

**Проблема**: При каждом подключении удаляются данные!

```javascript
// ❌ ОПАСНО!
con.connect().then(() => {
    console.log("Connected to the database");
}).catch(err => {
    console.error('Connection error', err.stack);
});

// Удаление дублей при КАЖДОМ подключении!
con.query(`
    DELETE FROM bookings
    WHERE id NOT IN (
        SELECT min_id FROM (
            SELECT MIN(id) as min_id
            FROM bookings
            GROUP BY user_id, resource_id
        ) AS keep
    )
`)
```

**Риск**:
- ❌ Потеря данных при каждом перезапуске
- ❌ Если несколько инстансов приложения - race condition
- ❌ Нет логирования удаленных записей

**Исправление**: Вынести в миграцию или отдельный скрипт
```javascript
// database/scripts/cleanup-duplicates.js
const con = require('./connection');

async function cleanupDuplicates() {
    try {
        const result = await con.query(`
            DELETE FROM bookings
            WHERE id NOT IN (
                SELECT min_id FROM (
                    SELECT MIN(id) as min_id
                    FROM bookings
                    GROUP BY user_id, resource_id
                ) AS keep
            )
            RETURNING *
        `);
        
        console.log(`Removed ${result.rowCount} duplicate bookings`);
        
        // Логирование удаленных записей
        result.rows.forEach(booking => {
            console.log(`Deleted booking: ${JSON.stringify(booking)}`);
        });
        
    } catch (err) {
        console.error('Error cleaning up duplicates:', err);
    }
}

// Запускать вручную или через cron, НЕ при подключении!
if (require.main === module) {
    cleanupDuplicates().then(() => process.exit(0));
}
```

    createLimiter,

### 2.3 Слабая валидация паролей

**Серьезность**: MEDIUM-HIGH  
**Файл**: `middleware/validators.js`  
**Строка**: 35-36

**Проблема**: Минимум 8 символов - слишком слабо

```javascript
// ❌ Слабая валидация
password: (password) => {
    return password && password.length >= 8;
}
```

**Исправление**: Усилить требования
```javascript
// ✅ Надежная валидация
password: (password) => {
    if (!password || password.length < 12) {
        return { valid: false, message: 'Пароль должен содержать минимум 12 символов' };
    }
    
    const hasUpperCase = /[A-Z]/.test(password);
    const hasLowerCase = /[a-z]/.test(password);
    const hasNumbers = /\d/.test(password);
    const hasSpecialChar = /[!@#$%^&*(),.?":{}|<>]/.test(password);
    
    if (!hasUpperCase) {
        return { valid: false, message: 'Пароль должен содержать заглавные буквы' };
    }
    if (!hasLowerCase) {
        return { valid: false, message: 'Пароль должен содержать строчные буквы' };
    }
    if (!hasNumbers) {
        return { valid: false, message: 'Пароль должен содержать цифры' };
    }
    if (!hasSpecialChar) {
        return { valid: false, message: 'Пароль должен содержать специальные символы' };
    }
    
    return { valid: true };
}

// Использование
const passwordValidation = validators.password(password);
if (!passwordValidation.valid) {
    return res.status(400).json({ error: passwordValidation.message });
}
```

```javascript

const errorHandler = (error, req, res, next) => {
    console.error(error);  // Может содержать чувствительные данные!
    
    const status = error.status || 500;
    const message = error.message || 'Internal Server Error';
    res.status(status).json({
        error: message,
        status: status,
        timestamp: new Date().toISOString()
    });
};
```

**Исправление**:
```javascript
const logger = require('../config/logger');

const errorHandler = (error, req, res, next) => {
    // Логировать только в файл с фильтрацией
    const sanitizedError = {
        message: error.message,
        stack: error.stack,
        status: error.status,
        // НЕ логировать req.body (может содержать пароли)
        url: req.url,
        method: req.method,
        ip: req.ip,
    };
    
    logger.error('Request failed', sanitizedError);
    
    const status = error.status || 500;
    
    // В production НЕ отправлять детали ошибок
    if (process.env.NODE_ENV === 'production' && status === 500) {
        return res.status(500).json({
            error: 'Internal Server Error',
            status: 500,
            timestamp: new Date().toISOString(),
        });
    }
    
    // В development можно больше деталей
    res.status(status).json({
        error: error.message,
        status: status,
        timestamp: new Date().toISOString(),
        ...(process.env.NODE_ENV === 'development' && { stack: error.stack }),
    });
};
```

## 5. РЕКОМЕНДАЦИИ OWASP

### Checklist безопасности:

- [ ] **A01: Broken Access Control**
  - ✅ Реализовать RBAC (Role-Based Access Control)
  - ✅ Проверять права на каждом эндпоинте
  - ✅ Не доверять client-side проверкам

- [ ] **A02: Cryptographic Failures**
  - ✅ Не использовать слабые алгоритмы
  - ✅ Хранить пароли только в хешированном виде (уже реализовано)
  - ✅ Использовать HTTPS

- [ ] **A03: Injection**
  - ✅ Использовать параметризованные запросы (частично реализовано)
  - ✅ Валидировать все входные данные
  - ✅ Использовать whitelist для названий таблиц

- [ ] **A04: Insecure Design**
  - ✅ Добавить rate limiting
  - ✅ Реализовать логирование безопасности
  - ✅ Добавить audit trail

- [ ] **A05: Security Misconfiguration**
  - ✅ Валидировать .env переменные
  - ✅ Не использовать дефолтные креды
  - ✅ Отключить подробные ошибки в production

- [ ] **A06: Vulnerable and Outdated Components**
  - ✅ Регулярно обновлять зависимости
  - ✅ Использовать `npm audit`
  - ✅ Не использовать deprecated пакеты
