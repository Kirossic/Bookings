// Файл: config/env.js
function validateEnv() {
    const required = ['DB_HOST', 'DB_USER', 'DB_PASSWORD', 'JWT_SECRET'];
    const missing = required.filter(key => !process.env[key]);
    
    if (missing.length > 0) {
        throw new Error(`❌ Missing: ${missing.join(', ')}`);
    }
    console.log('✅ Environment validated');
}
